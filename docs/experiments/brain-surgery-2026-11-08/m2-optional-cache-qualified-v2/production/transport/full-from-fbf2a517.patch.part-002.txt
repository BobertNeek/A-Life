+fn fresh_pair(
+    r: &recipe::Recipe,
+    preparation: u64,
+    sentinel: bool,
+    actual: &mut u64,
+    attempts: &Path,
+) -> Result<[Host; 2]> {
+    let seed = r.config.world_seed;
+    let game = newborn(r)?;
+    let second = game.clone();
+    let (full, template) = CpuGroupedM2HostV1::new_prepared(
+        game,
+        trainer(r, sentinel)?,
+        r.inputs.clone(),
+        sampling_seed(seed),
+        mode(0)?,
+        |world| -> Result<HeadlessWorld> {
+            for offset in 0..preparation {
+                write_new(
+                    attempts,
+                    &format!("BODY-ATTEMPT-{offset:06}.json"),
+                    &json!({"offset":offset,"next_tick":world.tick().raw()+1,"attempt_before_call":true,"no_retry":true}),
+                )?;
+                world.try_advance_tick()?;
+                *actual += 1;
+            }
+            if world.age_death_disabled()
+                || world.tick().raw() != preparation
+                || !world
+                    .organism_registry()
+                    .iter()
+                    .next()
+                    .ok_or("prepared body absent")?
+                    .lifecycle()
+                    .is_alive()
+            {
+                return Err("normal fresh preparation failed".into());
+            }
+            let evidence = scene::initial_audit(world)?;
+            write_new(attempts, "PREPARED-SCENE.json", &evidence)?;
+            Ok(world.clone())
+        },
+    )?;
+    let (sham, ()) = CpuOptionalVocalM2HostV1::new_prepared(
+        second,
+        trainer(r, sentinel)?,
+        r.inputs.clone(),
+        sampling_seed(seed),
+        mode(1)?,
+        |world| -> Result<()> {
+            *world = template;
+            Ok(())
+        },
+    )?;
+    let full = Host::Legacy(full);
+    let sham = Host::Optional(sham);
+    let a = cp(&full)?;
+    let b = cp(&sham)?;
+    raw::fresh(&a, preparation, sampling_seed(seed))?;
+    raw::fresh(&b, preparation, sampling_seed(seed))?;
+    if raw::matched_initial(&a, mode_name(0)?)? != raw::matched_initial(&b, mode_name(1)?)?
+        || *actual != preparation
+    {
+        return Err("fresh matched pair/preparation accounting differs".into());
+    }
+    Ok([full, sham])

+        || f.host_commit != HOST_COMMIT

+        || f.driver_commit != DRIVER_COMMIT

+    if hash(f.qualification_json.as_bytes()) != f.qualification_sha256
+        || q["complete"] != true

+        || q["driver_code_sha256"] != code_hash()
+        || q["executable_sha256"] != f.executable_sha256
+        || q["protocol_sha256"] != hash(PROTOCOL)
+        || q["actual_native_calls"] != 16

+        || q["primary_seed_executed"] != false
+        || q["observer_inertness_and_cold_exact"] != true

+        return Err("exact qualification missing or changed".into());
+    }
+    let max = q["maximum_packet_bytes"]
+        .as_u64()
+        .ok_or("qualification packet maximum absent")?;
+    if max
+        .checked_add(262144)
+        .is_none_or(|n| n > journal::MAX_PACKET)
+        || f.projected_pair_bytes != 2 * HORIZON * journal::MAX_PACKET

+    validate_activation(&claim, root, &hash(&fbytes), &f.reservation_sha256)?;
+    journal::publish_artifact(root, "FREEZE.json", &fbytes, 64 * 1024)?;
+    journal::publish_artifact(root, "RESERVATION.json", &rbytes, 1024 * 1024)?;
+    write_new(
+        root,
+        "PREPARATION_ATTEMPT.json",
+        &json!({"schema":"object-rich-optional-vocal-preparation-attempt-v1","source":DRIVER_COMMIT,"seed":SEED,"actual_steps_known_at_claim":0,"automatic_retry":false}),
+    )?;
+    let mut actual = 0;
+    let mut stage = "source_construction";
+    let mut source_completed = false;
+    let result = (|| -> Result<()> {
+        // Exactly one candidate is constructed, durably bound before any body step.
+        let r = recipe::build(SEED, true)?;
+        source_completed = true;
+        stage = "source_contract_before_biology";
+        let source_contract = source_contract(&r)?;
+        write_new(root, "SOURCE-CONTRACT.json", &source_contract)?;
+        source_completed = true;
+        stage = "source_binding_before_biology";
+        let source = serde_json::to_vec(&r.metadata)?;
+        let inputs = serde_json::to_vec(&r.inputs)?;
+        let source_bound = serde_json::to_vec(&json!({
+            "schema":"object-rich-optional-vocal-pre-preparation-source-v1",
+            "freeze_sha256":hash(&fbytes),"activation_sha256":hash(&claim_bytes),
+            "source_manifest_sha256":hash(&source),"inputs_sha256":hash(&inputs),
+            "genotype_sha256":r.metadata["genotype_sha256"],
+            "foundation_asset":r.metadata["target_asset"],
+            "actual_preparation_intervals":0,"native_calls":0,
+            "single_constructed_source_candidate":true
+        }))?;
+        let hosts = bind_then_prepare(root, &source, &inputs, &source_bound, || {
+            stage = "newborn_admission_and_shared_world_preparation";
+            fresh_pair(&r, PREPARATION, false, &mut actual, root)
+        })?;
+        stage = "initial_matched_pair_publication";
+        let mut arms = Vec::new();
+        let mut matched = None;
+        for (arm, h) in hosts.iter().enumerate() {
+            let bytes = cp(h)?;
+            let m = hash(&raw::matched_initial(&bytes, mode_name(arm)?)?);
+            if matched.as_ref().is_some_and(|old| old != &m) {
+                return Err("pair mismatch at publication".into());
+            }
+            matched = Some(m);
+            let d = root.join(format!("arm-{arm}"));
+            fs::create_dir(&d)?;
+            journal::publish_artifact(&d, "initial.private.m2", &bytes, journal::MAX_PACKET)?;
+            arms.push(PreparedArm {
+                arm,
+                mode: mode_name(arm)?.into(),
+                initial_sha256: hash(&bytes),
+                closure_sha256: closure_hash(&bytes)?,
+                source_headers: raw::source_headers(&bytes)?,
+            });
+        }
+        let p = Prepared {
+            schema: "object-rich-optional-vocal-preparation-v1".into(),
+            activation_sha256: hash(&claim_bytes),
+            pre_preparation_source_sha256: hash(&source_bound),
+            freeze_sha256: hash(&fbytes),
+            source_manifest_sha256: hash(&source),
+            inputs_sha256: hash(&inputs),
+            initial_matched_sha256: matched.ok_or("matched pair absent")?,
+            actual_preparation_intervals: actual,
+            initial_tick: PREPARATION,
+            initial_neural_decisions: 0,
+            sampling_seed: sampling_seed(SEED),
+            pressure_reference_bits: r
+                .phenotype
+                .cognitive_architecture()
+                .sleep_trigger_threshold()
+                .to_bits(),
+            arms,
+        };
+        let pbytes = serde_json::to_vec(&p)?;
+        // Complete preparation receipt is last; a partial tree cannot be run.
+        for arm in 0..2 {
+            journal::init_journal(
+                &root.join(format!("arm-{arm}/JOURNAL-M2-V1")),
+                &binding(&f, &p, &hash(&pbytes), arm)?,
+            )?;
+        }
+        journal::publish_artifact(root, "PREPARATION_COMPLETE.json", &pbytes, 64 * 1024)?;
+        Ok(())
+    })();
+    if let Err(e) = &result {
+        let _ = write_new(
+            root,
+            "PREPARATION_FAILED.json",
+            &json!({"error":e.to_string(),"failure_stage":stage,"source_construction_attempted":true,"source_construction_completed":source_completed,"actual_preparation_intervals":actual,"native_calls":0,"no_retry":true,"seed_consumed":actual>0}),
+        );
+    }
+    result
+}
+struct Loaded {
+    binding: journal::Binding,
+    inputs: PhenotypeCompilerInputs,
+    initial: Vec<u8>,
+    pressure_reference: f32,
+    journal: PathBuf,
+}
+fn load(root: &Path, arm: usize) -> Result<Loaded> {
+    mode(arm)?;
+    let (f, fbytes): (Freeze, _) = read_typed(&root.join("FREEZE.json"), 64 * 1024)?;
+    check_freeze(&f)?;
+    canonical_reservation(Path::new(RESERVATION_PATH))?;
+    let (claim, claim_bytes): (ActivationClaim, _) =
+        read_typed(Path::new(ACTIVATION_PATH), 64 * 1024)?;
+    validate_activation(&claim, root, &hash(&fbytes), &f.reservation_sha256)?;
+    if hash(&journal::bounded_read(
+        Path::new(RESERVATION_PATH),
+        1024 * 1024,
+    )?) != f.reservation_sha256
+    {
+        return Err("authoritative reserved source identity changed".into());
+    }
+    let (p, pbytes): (Prepared, _) =
+        read_typed(&root.join("PREPARATION_COMPLETE.json"), 64 * 1024)?;
+    if p.schema != "object-rich-optional-vocal-preparation-v1"
+        || p.activation_sha256 != hash(&claim_bytes)
+        || p.freeze_sha256 != hash(&fbytes)
+        || p.actual_preparation_intervals != PREPARATION
+        || p.initial_tick != PREPARATION
+        || p.initial_neural_decisions != 0
+        || p.sampling_seed != sampling_seed(SEED)
+        || p.arms.len() != 2
+        || root.join("PREPARATION_FAILED.json").exists()
+    {
+        return Err("incomplete or changed preparation".into());
+    }
+    let (source_bound, source_bound_bytes): (Value, _) = read_typed(
+        &root.join("SOURCE_BOUND_BEFORE_PREPARATION.json"),
+        64 * 1024,
+    )?;
+    if hash(&source_bound_bytes) != p.pre_preparation_source_sha256
+        || source_bound["schema"] != "object-rich-optional-vocal-pre-preparation-source-v1"
+        || source_bound["freeze_sha256"] != p.freeze_sha256
+        || source_bound["activation_sha256"] != p.activation_sha256
+        || source_bound["source_manifest_sha256"] != p.source_manifest_sha256
+        || source_bound["inputs_sha256"] != p.inputs_sha256
+        || source_bound["actual_preparation_intervals"] != 0
+        || source_bound["native_calls"] != 0
+        || source_bound["single_constructed_source_candidate"] != true
+    {
+        return Err("pre-preparation source identity differs".into());
+    }
+    if hash(&journal::bounded_read(
+        &root.join("SOURCE.json"),
+        journal::MAX_PACKET,
+    )?) != p.source_manifest_sha256
+    {
+        return Err("source metadata changed".into());
+    }
+    let (inputs, ibytes): (PhenotypeCompilerInputs, _) =
+        read_typed(&root.join("INPUTS.private.json"), journal::MAX_PACKET)?;
+    if hash(&ibytes) != p.inputs_sha256 {
+        return Err("source input bytes changed".into());
+    }
+    let b = binding(&f, &p, &hash(&pbytes), arm)?;
+    let initial = journal::bounded_read(
+        &root.join(format!("arm-{arm}/initial.private.m2")),
+        journal::MAX_PACKET,
+    )?;
+    if hash(&initial) != b.initial_sha256
+        || hash(&raw::matched_initial(&initial, mode_name(arm)?)?) != b.initial_matched_sha256
+        || raw::source_headers(&initial)? != b.source_headers
+    {
+        return Err("initial source/matched bytes changed".into());
+    }
+    raw::fresh(&initial, PREPARATION, b.sampling_seed)?;
+    validate_cp(&initial, b.initial_tick, &b.closure_sha256)?;
+    let pressure = inputs
+        .genome()
+        .cognitive_architecture()
+        .sleep_trigger_threshold();
+    if pressure.to_bits() != p.pressure_reference_bits {
+        return Err("compiled pressure reference changed".into());
+    }
+    Ok(Loaded {
+        binding: b,
+        inputs,
+        initial,
+        pressure_reference: pressure,
+        journal: root
+            .canonicalize()?
+            .join(format!("arm-{arm}/JOURNAL-M2-V1")),
+    })
+}
+fn run(root: &Path, arm: usize) -> Result<()> {
+    let a = load(root, arm)?;
+    let b = &a.binding;
+    let _lock = journal::lock(&a.journal)?;
+    let mut r = journal::recover_without_trace(&a.journal, b, &a.initial, |bytes, tick| {
+        validate_cp(bytes, tick, &b.closure_sha256)
+    })?;
+    if a.journal.join("BLOCKED-PUBLICATION").exists() {
+        return Err("known failure blocks continuation".into());
+    }
+    if a.journal.join("TERMINAL.json").exists() {
+        println!("terminal no-op; {} committed intervals", r.next);
+        return Ok(());
+    }
+    let result = (|| -> Result<()> {
+        let mut h = restore(
+            &r.checkpoint,
+            &a.inputs,
+            arm,
+            b.initial_tick + r.next,
+            &b.closure_sha256,
+        )?;
+        if r.trace_bytes
+            .checked_add(b.max_packet_bytes)
+            .is_none_or(|n| n > journal::MAX_TRACE_BYTES)
+            || journal::resource_stop(b, r.storage_bytes, journal::free_bytes(&a.journal)?)
+        {
+            journal::terminal(
+                &a.journal,
+                b,
+                &r,
+                "pre-step storage or science-trace ceiling",
+            )?;
+            return Ok(());
+        }
+        write_new(
+            &a.journal,
+            &format!(
+                "PROCESS-{}-{}.json",
+                std::process::id(),
+                std::time::SystemTime::now()
+                    .duration_since(std::time::UNIX_EPOCH)?
+                    .as_nanos()
+            ),
+            &json!({"schema":"object-rich-optional-vocal-process-v1","arm":arm,"next_offset":r.next,"previous_commit_sha256":r.previous,"process_id":std::process::id(),"retained_temp_entries":r.ignored_temps,"unpublished_inflight_work_is_uncertain":true,"published_intervals_replayed":0}),
+        )?;
+        r.storage_bytes = fs::read_dir(&a.journal)?.try_fold(0_u64, |sum, e| -> Result<u64> {
+            sum.checked_add(e?.metadata()?.len())
+                .ok_or_else(|| "storage overflow".into())
+        })?;
+        while r.next < b.horizon {
+            if r.trace_bytes
+                .checked_add(b.max_packet_bytes)
+                .is_none_or(|n| n > journal::MAX_TRACE_BYTES)
+                || journal::resource_stop(b, r.storage_bytes, journal::free_bytes(&a.journal)?)
+            {
+                journal::terminal(
+                    &a.journal,
+                    b,
+                    &r,
+                    "pre-step storage or science-trace ceiling",
+                )?;
+                return Ok(());
+            }
+            let before = observe(&h)?;
+            if !before.alive
+                || before.body.health < 0.5
+                || before.body.energy < 0.05
+                || before.body.injury > 0.
+                || !before.body.health.is_finite()
+                || !before.body.energy.is_finite()
+                || !before.body.injury.is_finite()
+            {
+                journal::terminal(&a.journal, b, &r, "health/energy/death")?;
+                return Ok(());
+            }
+            let attempt_name = format!("ATTEMPT-{:06}.json", r.next);
+            if a.journal.join(&attempt_name).exists() {
+                return Err("unresolved attempted interval cannot replay".into());
+            }
+            write_new(
+                &a.journal,
+                &attempt_name,
+                &json!({"offset":r.next,"tick":before.tick,"previous_commit_sha256":r.previous,"attempt_before_call":true,"no_retry":true}),
+            )?;
+            r.storage_bytes = r
+                .storage_bytes
+                .checked_add(2 * fs::metadata(a.journal.join(&attempt_name))?.len())
+                .ok_or("attempt storage overflow")?;
+            let produced = (|| -> Result<(Vec<u8>, Vec<u8>, u64, bool)> {
+                let step = h.step(f32::from_bits(b.temperature_bits))?;
+                let bytes = cp(&h)?;
+                validate_cp(&bytes, b.initial_tick + r.next + 1, &b.closure_sha256)?;
+                let rows = science(
+                    &before,
+                    &observe(&h)?,
+                    &step,
+                    r.next,
+                    &bytes,
+                    &b.closure_sha256,
+                    a.pressure_reference,
+                )?;
+                let packet = journal::make_packet(b, r.next, &r.previous, &bytes, &rows)?;
+                journal::decode_packet(&packet, b, r.next, &r.previous)?;
+                let fresh_harm = step.decisions.iter().any(|d| {
+                    d["research_world_receipt"]["body_event"]["damage"]
+                        .as_f64()
+                        .is_some_and(|v| v > 0.)
+                });
+                Ok((bytes, packet, rows.len() as u64, fresh_harm))
+            })();
+            let (bytes, packet, trace_bytes, fresh_harm) = match produced {
+                Ok(v) => v,
+                Err(e) => {
+                    journal::stop_after_failure(
+                        &a.journal,
+                        b,
+                        &r,
+                        &format!("step/audit/serialization failure: {e}"),
+                    );
+                    return Err(e);
+                }
+            };
+            if let Err(e) = journal::publish(&a.journal, &format!("{:06}.packet", r.next), &packet)
+            {
+                let reason = format!("known publication/storage failure: {e}");
+                journal::record_publication_failure(
+                    &a.journal, b, &mut r, &bytes, &packet, &reason,
+                )?;
+                return Err(reason.into());
+            }
+            r.trace_bytes = r
+                .trace_bytes
+                .checked_add(trace_bytes)
+                .ok_or("trace accounting overflow")?;
+            r.next += 1;
+            r.previous = hash(&packet);
+            r.checkpoint = bytes;
+            r.storage_bytes = r
+                .storage_bytes
+                .checked_add(2 * packet.len() as u64)
+                .ok_or("storage overflow after commit")?;
+            if fresh_harm {
+                journal::terminal(&a.journal, b, &r, "new measured physical damage")?;
+                return Ok(());
+            }
+            if r.next.is_multiple_of(16) {
+                println!("arm {arm}: {} atomically committed intervals", r.next);
+            }
+        }
+        journal::terminal(&a.journal, b, &r, "horizon")
+    })();
+    if let Err(error) = &result {
+        if !a.journal.join("TERMINAL.json").exists()
+            && !a.journal.join("BLOCKED-PUBLICATION").exists()
+        {
+            journal::stop_after_failure(
+                &a.journal,
+                b,
+                &r,
+                &format!("known driver/storage failure: {error}"),
+            );
+        }
+    }
+    result
+}
+fn export(root: &Path, arm: usize, out: &Path) -> Result<()> {
+    let a = load(root, arm)?;
+    let _lock = journal::lock(&a.journal)?;
+    let b = &a.binding;
+    let r = journal::recover(&a.journal, b, &a.initial, |bytes, tick| {
+        validate_cp(bytes, tick, &b.closure_sha256)
+    })?;
+    let h = restore(
+        &r.checkpoint,
+        &a.inputs,
+        arm,
+        b.initial_tick + r.next,
+        &b.closure_sha256,
+    )?;
+    let terminal = if a.journal.join("TERMINAL.json").is_file() {
+        Some(read_typed::<Value>(&a.journal.join("TERMINAL.json"), 64 * 1024)?.0)
+    } else {
+        None
+    };
+    let summary = summarize(
+        &r.trace,
+        &observe(&h)?,
+        r.next,
+        terminal.as_ref(),
+        &r.previous,
+    )?;
+    fs::create_dir(out)?;
+    journal::publish_artifact(out, "TRACE.private.jsonl", &r.trace, journal::MAX_ARM_BYTES)?;
+    journal::publish_artifact(out, "final.private.m2", &r.checkpoint, journal::MAX_PACKET)?;
+    let result = serde_json::to_vec(&summary)?;
+    journal::publish_artifact(out, "RESULT.json", &result, journal::MAX_PACKET)?;
+    write_new(
+        out,
+        "EXPORT_COMPLETE.json",
+        &json!({"schema":"object-rich-optional-vocal-export-v1","binding":b,"committed_intervals":r.next,"terminal":terminal.is_some(),"last_commit_sha256":r.previous,"trace_sha256":hash(&r.trace),"checkpoint_sha256":hash(&r.checkpoint),"result_sha256":hash(&result),"private_raw_data_not_for_upload":true}),
+    )

+#[derive(Debug, Serialize)]
+struct CycleProgress {
+    start_tick: u64,
+    entering_tick: Option<u64>,
+    consolidating_tick: Option<u64>,
+    waking_tick: Option<u64>,
+    ordered: bool,
+}
+impl CycleProgress {
+    fn new(start_tick: u64, first_phase: SleepPhase, observed_tick: u64) -> Self {
+        Self {
+            start_tick,
+            entering_tick: (first_phase == SleepPhase::EnteringSleep).then_some(observed_tick),
+            consolidating_tick: None,
+            waking_tick: None,
+            ordered: first_phase == SleepPhase::EnteringSleep,
+        }
+    }
+    fn observe(&mut self, phase: SleepPhase, tick: u64) {
+        match phase {
+            SleepPhase::EnteringSleep
+                if self.consolidating_tick.is_none() && self.waking_tick.is_none() => {}
+            SleepPhase::Consolidating
+                if self.entering_tick.is_some() && self.waking_tick.is_none() =>
+            {
+                self.consolidating_tick.get_or_insert(tick);
+            }
+            SleepPhase::Waking if self.consolidating_tick.is_some() => {
+                self.waking_tick.get_or_insert(tick);
+            }
+            _ => self.ordered = false,
+        }
+    }
+    fn complete(&self) -> bool {
+        self.ordered
+            && self.entering_tick.is_some()
+            && self.consolidating_tick.is_some()
+            && self.waking_tick.is_some()
+    }
+}
+fn summarize(
+    trace: &[u8],
+    final_state: &Observation,
+    count: u64,
+    terminal: Option<&Value>,
+    last: &str,
+) -> Result<Value> {
+    let mut decisions = 0_u64;
+    let mut physical = [0_u64; 3];
+    let mut attempted = [0_u64; 3];
+    let mut executed_noops = 0_u64;
+    let mut no_decision = 0_u64;
+    let mut consolidations = 0_u64;
+    let mut phase_counts = BTreeMap::<String, u64>::new();
+    let mut crosses = [[0_u64; 2]; 2];
+    let mut observed = 0;
+    let mut started_cycles = BTreeMap::<u64, CycleProgress>::new();
+    let mut completed_cycles = Vec::new();
+    let mut sleep_starts = Vec::new();
+    let mut crossing_events = Vec::new();
+    let mut first_up = [None, None];
+    let mut maxima = [f32::NEG_INFINITY; 2];
+    let mut minimum_health = f32::INFINITY;
+    let mut minimum_energy = f32::INFINITY;
+    let mut initially_at_or_above_reference = None;
+    for line in trace.split(|b| *b == b'\n').filter(|l| !l.is_empty()) {
+        let v: Value = serde_json::from_slice(line)?;
+        if v["status"] == "committed" {
+            let rows = v["decisions"].as_array().ok_or("decisions missing")?;
+            decisions += rows.len() as u64;
+            if rows.is_empty() {
+                no_decision += 1;
+            }
+            consolidations += v["consolidations"]
+                .as_u64()
+                .ok_or("consolidations missing")?;
+            for d in rows {
+                for (value, counts) in [
+                    (&d["physical"]["operation"], &mut attempted),
+                    (&d["applied_recovery"], &mut physical),
+                ] {
+                    match value.as_str() {
+                        Some("AwakeRest") => counts[0] += 1,
+                        Some("BodySleep") => counts[1] += 1,
+                        None => counts[2] += 1,
+                        _ => return Err("unknown recovery operation".into()),
+                    }
+                }
+                for noop in d["real_noop_receipts"]
+                    .as_array()
+                    .ok_or("no-op receipts absent")?
+                {
+                    if noop["measured"]["executed"] == true {
+                        executed_noops += 1;
+                    }
+                }
+            }
+        } else if v["status"] == "observed" {
+            observed += 1;
+            let before: Observation = serde_json::from_value(v["before"].clone())?;
+            let after: Observation = serde_json::from_value(v["after"].clone())?;
+            if before.phase == SleepPhase::Awake && after.phase != SleepPhase::Awake {
+                let cycle = after.sleep.active_cycle_id;
+                started_cycles.insert(
+                    cycle,
+                    CycleProgress::new(before.tick, after.phase, after.tick),
+                );
+                sleep_starts.push(json!({"cycle":cycle,"tick":before.tick,"trigger":after.sleep.last_trigger,"phase":after.phase}));
+            }
+            if let Some(progress) = started_cycles.get_mut(&after.sleep.active_cycle_id) {
+                progress.observe(after.phase, after.tick);
+            }
+            if before.phase == SleepPhase::Waking && after.phase == SleepPhase::Awake {
+                let cycle = before.sleep.active_cycle_id;
+                if started_cycles
+                    .get(&cycle)
+                    .is_some_and(CycleProgress::complete)
+                {
+                    let progress = started_cycles.remove(&cycle).ok_or("cycle disappeared")?;
+                    completed_cycles.push(json!({"cycle":cycle,"start_tick":progress.start_tick,"entering_observed_tick":progress.entering_tick,"consolidating_observed_tick":progress.consolidating_tick,"waking_observed_tick":progress.waking_tick,"returned_awake_tick":after.tick}));
+                }
+            }
+            *phase_counts
+                .entry(format!("{:?}", after.phase))
+                .or_default() += 1;
+            let pressure = f32::from_bits(
+                v["pressure_reference_bits"]
+                    .as_u64()
+                    .ok_or("pressure bits absent")? as u32,
+            );
+            minimum_health = minimum_health
+                .min(before.body.health)
+                .min(after.body.health);
+            minimum_energy = minimum_energy
+                .min(before.body.energy)
+                .min(after.body.energy);
+            if initially_at_or_above_reference.is_none() {
+                initially_at_or_above_reference = Some([
+                    before.homeostasis.drives.fatigue >= 0.35,
+                    before.homeostasis.hormones.sleep_pressure >= pressure,
+                ]);
+            }
+            for (axis, (pre, post, threshold)) in [
+                (
+                    before.homeostasis.drives.fatigue,
+                    after.homeostasis.drives.fatigue,
+                    0.35_f32,
+                ),
+                (
+                    before.homeostasis.hormones.sleep_pressure,
+                    after.homeostasis.hormones.sleep_pressure,
+                    pressure,
+                ),
+            ]
+            .into_iter()
+            .enumerate()
+            {
+                maxima[axis] = maxima[axis].max(pre).max(post);
+                if pre < threshold && post >= threshold {
+                    crosses[axis][0] += 1;
+                    first_up[axis].get_or_insert(after.tick);
+                    crossing_events.push(json!({"axis":if axis==0{"fatigue"}else{"sleep_pressure"},"direction":"up","tick":after.tick,"before":pre,"after":post,"reference_bits":threshold.to_bits()}));
+                }
+                if pre >= threshold && post < threshold {
+                    crosses[axis][1] += 1;
+                    crossing_events.push(json!({"axis":if axis==0{"fatigue"}else{"sleep_pressure"},"direction":"down","tick":after.tick,"before":pre,"after":post,"reference_bits":threshold.to_bits()}));
+                }
+            }
+        } else {
+            return Err("unknown science row".into());
+        }
+    }
+    if observed != count {
+        return Err("summary interval count differs".into());
+    }
+    if count == 0 {

+        return Err("fresh grouped-M2 guard claim differs from manifest".into());
+    }
+    validate_cp(initial, binding.initial_tick)?;
+    let mut names = Vec::new();
+    let mut entry_count = 0_u64;
+    let mut storage_bytes = 0_u64;
+    let mut ignored_temps = 0;
+    // Charge every directory entry conservatively, even retained hardlink aliases.
+    for entry in fs::read_dir(dir)? {
+        let entry = entry?;
+        entry_count += 1;
+        if entry_count > binding.horizon.saturating_mul(4).saturating_add(128) {
+            return Err("journal entry count exceeds bounded horizon allowance".into());
+        }
+        let name = entry
+            .file_name()
+            .into_string()
+            .map_err(|_| "non-UTF8 journal name")?;
+        let meta = fs::symlink_metadata(entry.path())?;
+        if !meta.is_file() || meta.file_type().is_symlink() {
+            return Err("journal contains nonregular entry".into());
+        }
+        if !allowed_name(&name)
+            || meta.len()
+                > if packet_name(&name) || name.starts_with(".tmp-") {
+                    binding.max_packet_bytes
+                } else {
+                    MAX_METADATA
+                }
+        {
+            return Err(format!("unknown or oversized journal entry: {name}").into());
+        }
+        storage_bytes = storage_bytes
+            .checked_add(meta.len())
+            .ok_or("storage overflow")?;
+        if name.starts_with(".tmp-") {
+            ignored_temps += 1;
+            continue;
+        }
+        if name.ends_with(".packet") {
+            names.push(name);
+        } else if name != "CLAIM.json"
+            && name != "MANIFEST.json"
+            && name != "LOCK"
+            && name != "BLOCKED-PUBLICATION"
+            && !name.starts_with("PROCESS-")
+            && !attempt_name(&name)
+            && name != "TERMINAL.json"
+        {
+            return Err(format!("unknown journal entry: {name}").into());
+        }
+    }
+    if storage_bytes > binding.max_arm_bytes {
+        return Err("journal exceeds storage ceiling".into());
+    }
+    bounded_read(&dir.join("LOCK"), 0)?;
+    names.sort();
+    let mut r = Recovery {
+        next: 0,
+        previous: hash(initial),
+        checkpoint: initial.to_vec(),
+        trace: Vec::new(),
+        trace_bytes: 0,
+        storage_bytes,
+        ignored_temps,
+    };
+    for name in names {
+        if name != format!("{:06}.packet", r.next) {
+            return Err("duplicate, omitted or out-of-order committed packet".into());
+        }
+        let wire = bounded_read(&dir.join(name), binding.max_packet_bytes)?;
+        let p = decode_packet(&wire, binding, r.next, &r.previous)?;
+        validate_cp(&p.checkpoint, p.header.tick_after)?;
+        // Every valid prefix must remain exportable, including no-trace resume.
+        r.trace_bytes = checked_trace_len(r.trace_bytes, p.trace.len() as u64)?;
+        r.next += 1;
+        r.previous = p.digest;
+        r.checkpoint = p.checkpoint;
+        if retain_trace {
+            append_trace(&mut r.trace, &p.trace)?;
+        }
+    }
+    if dir.join("TERMINAL.json").exists() {
+        let t: Value =
+            serde_json::from_slice(&bounded_read(&dir.join("TERMINAL.json"), 16 * 1024)?)?;
+        if t["schema"] != "object-rich-optional-vocal-terminal-v1"
+            || t["binding"] != serde_json::to_value(binding)?
+            || t["committed_intervals"] != r.next
+            || t["last_commit_sha256"] != r.previous
+            || t["final_checkpoint_sha256"] != hash(&r.checkpoint)
+            || t["tick"] != binding.initial_tick + r.next
+            || !t["stop"].is_string()
+        {
+            return Err("terminal record differs from complete committed prefix".into());
+        }
+    }
+    Ok(r)
+}
+pub fn lock(dir: &Path) -> Result<File> {
+    regular_dir(dir)?;
+    bounded_read(&dir.join("LOCK"), 0)?;
+    let f = OpenOptions::new()
+        .read(true)
+        .write(true)
+        .open(dir.join("LOCK"))?;
+    f.try_lock()
+        .map_err(|e| format!("journal already has an active writer: {e}"))?;
+    Ok(f)
+}
+pub fn free_bytes(dir: &Path) -> Result<u64> {
+    // POSIX df reports available blocks for this exact filesystem. Fail closed
+    // on missing utility or unexpected output; no fragile ABI-specific statvfs.
+    let out = std::process::Command::new("df")
+        .args(["-Pk", "--"])
+        .arg(dir)
+        .env("LC_ALL", "C")
+        .output()?;
+    if !out.status.success() {
+        return Err("free-space inspection failed".into());
+    }
+    let text = String::from_utf8(out.stdout)?;
+    let row = text
+        .lines()
+        .last()
+        .ok_or("missing df output")?
+        .split_whitespace()
+        .collect::<Vec<_>>();
+    if row.len() < 6 {
+        return Err("invalid df output".into());
+    }
+    Ok(row[3]
+        .parse::<u64>()?
+        .checked_mul(1024)
+        .ok_or("df overflow")?)
+}
+pub fn resource_stop(b: &Binding, used: u64, free: u64) -> bool {
+    validate_binding(b).is_err()
+        || b.max_packet_bytes
+            .checked_mul(2)
+            .and_then(|n| n.checked_add(METADATA_RESERVE))
+            .and_then(|n| used.checked_add(n))
+            .is_none_or(|n| n > b.max_arm_bytes)
+        || b.min_free_bytes
+            .checked_add(b.max_packet_bytes)
+            .is_none_or(|n| free < n)
+}
+pub fn terminal(dir: &Path, b: &Binding, r: &Recovery, reason: &str) -> Result<()> {
+    validate_binding(b)?;
+    if r.next > b.horizon {
+        return Err("terminal exceeds horizon".into());
+    }
+    publish(
+        dir,
+        "TERMINAL.json",
+        &serde_json::to_vec(
+            &json!({"schema":"object-rich-optional-vocal-terminal-v1","binding":b,"stop":reason,"committed_intervals":r.next,"last_commit_sha256":r.previous,"final_checkpoint_sha256":hash(&r.checkpoint),"tick":b.initial_tick+r.next}),
+        )?,
+    )
+}
+pub fn stop_after_failure(dir: &Path, b: &Binding, r: &Recovery, reason: &str) {
+    if let Err(stop_error) = terminal(dir, b, r, reason) {
+        let blocker = OpenOptions::new()
+            .create_new(true)
+            .write(true)
+            .open(dir.join("BLOCKED-PUBLICATION"));
+        if let Ok(f) = blocker {
+            let _ = f.sync_all();
+            let _ = sync_dir(dir);
+        }
+        eprintln!("STOP RECORD FAILURE: {stop_error}. Do not resume automatically; storage may be unable to persist the blocker. {reason}");
+    }
+}
+pub fn record_publication_failure(
+    dir: &Path,
+    b: &Binding,
+    r: &mut Recovery,
+    cp: &[u8],
+    packet: &[u8],
+    reason: &str,
+) -> Result<()> {
+    let decoded = match decode_packet(packet, b, r.next, &r.previous) {
+        Ok(p) if p.checkpoint == cp => p,
+        _ => {
+            stop_after_failure(
+                dir,
+                b,
+                r,
+                "invalid failed-publication packet; journal blocked",
+            );
+            return Err("invalid failed-publication packet".into());
+        }
+    };
+    let mut trace_failure = None;
+    let published = dir.join(format!("{:06}.packet", r.next));
+    match fs::symlink_metadata(&published) {
+        Ok(_) => {
+            match bounded_read(&published, b.max_packet_bytes) {
+                Ok(actual) if actual == packet => {}
+                _ => {
+                    stop_after_failure(
+                        dir,
+                        b,
+                        r,
+                        "conflicting publication destination; journal invalid",
+                    );
+                    return Err("publication destination conflicts or cannot be read".into());
+                }
+            }
+            // A successful link is committed even when its directory sync failed.
+            // Reconcile before emitting any terminal status; never replay it.
+            r.next += 1;
+            r.previous = decoded.digest;
+            r.checkpoint = decoded.checkpoint;
+            // Update durable prefix identity first. Aggregate memory exhaustion
+            // must never erase a known-visible commit or cause its replay.
+            match r.trace_bytes.checked_add(decoded.trace.len() as u64) {
+                Some(total) => {
+                    // Preserve the exact visible byte count even if an invalid
+                    // caller allowed a publication beyond the export ceiling.
+                    r.trace_bytes = total;
+                    if total > MAX_TRACE_BYTES {
+                        trace_failure =
+                            Some("raw science exceeds 512 MiB aggregate memory cap".into());
+                    }
+                }
+                None => trace_failure = Some("raw science length overflow".into()),
+            }
+            if trace_failure.is_none() {
+                if let Err(error) = append_trace(&mut r.trace, &decoded.trace) {
+                    trace_failure = Some(error.to_string());
+                }
+            }
+        }
+        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {}
+        Err(e) => {
+            stop_after_failure(
+                dir,
+                b,
+                r,
+                "publication outcome cannot be inspected; journal blocked",
+            );
+            return Err(e.into());
+        }
+    }
+    let stop_reason = match &trace_failure {
+        Some(error) => format!("{reason}; raw science aggregation stopped: {error}"),
+        None => reason.to_owned(),
+    };
+    stop_after_failure(dir, b, r, &stop_reason);
+    // Count what actually exists, including a prelink temporary, a successful
+    // link's retained alias, and all terminal/blocker publication evidence.
+    // Do not assume a failed write produced zero bytes or exactly one temp.
+    let (storage_bytes, ignored_temps) = storage_snapshot(dir, b)?;
+    r.storage_bytes = storage_bytes;
+    r.ignored_temps = ignored_temps;
+    if let Some(error) = trace_failure {
+        return Err(error.into());
+    }
+    Ok(())
+}
+
+fn checked_trace_len(current: u64, additional: u64) -> Result<u64> {
+    let total = current
+        .checked_add(additional)
+        .ok_or("raw science length overflow")?;
+    if total > MAX_TRACE_BYTES {
+        return Err("raw science exceeds 512 MiB aggregate memory cap".into());
+    }
+    Ok(total)
+}
+fn append_trace(trace: &mut Vec<u8>, incoming: &[u8]) -> Result<()> {
+    checked_trace_len(trace.len() as u64, incoming.len() as u64)?;
+    // Bound before requesting allocation; avoid Vec's speculative growth factor.
+    trace.try_reserve_exact(incoming.len())?;
+    trace.extend_from_slice(incoming);
+    Ok(())
+}
+
+fn storage_snapshot(dir: &Path, b: &Binding) -> Result<(u64, usize)> {
+    regular_dir(dir)?;
+    let mut storage_bytes = 0_u64;
+    let mut ignored_temps = 0;
+    for (index, entry) in fs::read_dir(dir)?.enumerate() {
+        if index as u64 >= b.horizon.saturating_mul(4).saturating_add(128) {
+            return Err("journal entry count exceeds bounded horizon allowance".into());
+        }
+        let entry = entry?;
+        let name = entry
+            .file_name()
+            .into_string()
+            .map_err(|_| "non-UTF8 journal name")?;
+        let meta = fs::symlink_metadata(entry.path())?;
+        if !meta.file_type().is_file() || !allowed_name(&name) {
+            return Err("cannot account for nonregular or unknown journal entry".into());
+        }
+        let limit = if packet_name(&name) || name.starts_with(".tmp-") {
+            b.max_packet_bytes
+        } else {
+            MAX_METADATA
+        };
+        if meta.len() > limit {
+            return Err("oversized journal entry".into());
+        }
+        storage_bytes = storage_bytes
+            .checked_add(meta.len())
+            .ok_or("storage overflow")?;
+        if name.starts_with(".tmp-") {
+            ignored_temps += 1;
+        }
+    }
+    Ok((storage_bytes, ignored_temps))
+}
+
+fn regular_dir(dir: &Path) -> Result<()> {
+    if !fs::symlink_metadata(dir)?.file_type().is_dir() {
+        return Err("journal path is not a real directory".into());
+    }
+    Ok(())
+}
+fn packet_name(name: &str) -> bool {
+    name.len() == 13
+        && name.ends_with(".packet")
+        && name.as_bytes()[..6].iter().all(u8::is_ascii_digit)
+}
+fn attempt_name(name: &str) -> bool {
+    name.len() == 19
+        && name.starts_with("ATTEMPT-")
+        && name.ends_with(".json")
+        && name.as_bytes()[8..14].iter().all(u8::is_ascii_digit)
+}
+fn allowed_name(name: &str) -> bool {
+    if name.contains('/') || name.contains('\\') || name.len() > 128 {
+        return false;
+    }
+    matches!(
+        name,
+        "MANIFEST.json" | "LOCK" | "CLAIM.json" | "BLOCKED-PUBLICATION" | "TERMINAL.json"
+    ) || packet_name(name)
+        || (name.starts_with("PROCESS-") && name.len() > 8)
+        || attempt_name(name)
+        || (name.starts_with(".tmp-") && name.len() > 5)
+}
+fn validate_binding(b: &Binding) -> Result<()> {
+    if b.schema != "object-rich-optional-vocal-durable-v1"
+        || b.study_id.is_empty()
+        || b.horizon == 0
+        || b.horizon > 1_000_000
+        || b.initial_tick.checked_add(b.horizon).is_none()
+        || b.max_packet_bytes == 0
+        || b.max_packet_bytes > MAX_PACKET
+        || b.max_arm_bytes == 0
+        || b.max_arm_bytes > MAX_ARM_BYTES
+        || b.min_free_bytes < MIN_FREE_BYTES
+        || serde_json::to_vec(b)?.len() as u64 > MAX_METADATA
+    {
+        return Err("invalid grouped-M2 binding or resource bounds".into());
+    }
+    Ok(())
+}
+/// Fresh directory only. The claim is guard-only identity: it does not establish
+/// a process start, a scientific decision, or an executed world interval.
+/// Interrupted initialization is intentionally not repaired during resume.
+pub fn init_journal(dir: &Path, b: &Binding) -> Result<()> {
+    validate_binding(b)?;
+    let manifest = serde_json::to_vec(b)?;
+    fs::create_dir(dir)?;
+    sync_dir(dir.parent().ok_or("journal parent missing")?)?;
+    let f = OpenOptions::new()
+        .create_new(true)
+        .write(true)
+        .open(dir.join("LOCK"))?;
+    f.sync_all()?;
+    publish(dir, "MANIFEST.json", &manifest)?;
+    // CREATE-NEW hardlink preserves any unknown conflicting claim unchanged.
+    fs::hard_link(dir.join("MANIFEST.json"), dir.join("CLAIM.json"))?;
+    sync_dir(dir)?;
+    Ok(())
+}
