+/// The only differing bytes allowed are envelope identity, exact outer policy
+/// markers, the two exact schema markers, the fresh optional owner object and
+/// the policy-derived organism brain digest. Full mode is NOT masked.
+/// Every other subtree is included, even fields unknown to this helper. Scalar
+/// tokens are copied verbatim, never deserialized as floats or re-emitted.
+/// Object keys are sorted; arrays retain order and length. Length framing avoids
+/// ambiguous concatenation. This is a comparison encoding, NOT a checkpoint.
+pub fn matched_initial(cp: &[u8], policy_name: &str) -> Result<Vec<u8>> {
+    let optional = match policy_name {
+        "LegacyGroupedM2/Full" => false,
+        "OptionalVocalM2/Full" => true,
+        _ => return Err("unknown initial policy name".into()),
+    };
+    if cp.starts_with(OPTIONAL_MAGIC) != optional {
+        return Err("initial envelope policy mismatch".into());
+    }
+    let p = payload(cp)?;
+    exact_string(
+        p.field("source_lineage")?,
+        if optional {
+            OPTIONAL_LINEAGE
+        } else {
+            LEGACY_LINEAGE
+        },
+    )?;
+    exact_string(
+        p.field("policy")?,
+        if optional {
+            OPTIONAL_POLICY
+        } else {
+            LEGACY_POLICY
+        },
+    )?;
+    exact_string(p.field("mode")?, "Full")?;
+    if p.field("schema_version")?.value::<u16>()? != 1 {
+        return Err("wrong outer schema".into());
+    }
+    let canonical = p.field("canonical")?;
+    if canonical.field("schema_version")?.value::<u16>()? != if optional { 0x4f56 } else { 2 } {
+        return Err("wrong canonical policy schema".into());
+    }
+    let neural = canonical.field("residents")?.one()?.field("neural")?;
+    if neural.field("schema_version")?.value::<u32>()? != if optional { 0x4f56_0001 } else { 2 } {
+        return Err("wrong neural policy schema".into());
+    }
+    let controllers = p.field("controllers")?.object()?;
+    if controllers.len() != 1 {
+        return Err("initial controller roster differs".into());
+    }
+    let (owner, controller) = controllers.iter().next().ok_or("missing controller")?;
+    // Exact resident ownership is also enforced by typed restore; reject any
+    // controller key not naming the one neural resident in this projection.
+    if owner != &neural.field("organism_id")?.value::<u64>()?.to_string() {
+        return Err("initial controller owner differs".into());
+    }
+    exact_string(controller.path(&["motor", "mode"])?, "Full")?;
+    let fields = controller.object()?;
+    match (optional, fields.get("optional_vocal")) {
+        (false, None) => (),
+        (true, Some(marker)) => {
+            let marker = marker.object()?;
+            if marker.len() != 2 || !marker.contains_key("schema") || !marker.contains_key("last") {
+                return Err("unexpected optional owner fields".into());
+            }
+            exact_string(marker["schema"], OPTIONAL_OWNER)?;
+            if marker["last"].0 != b"null" {
+                return Err("optional owner is not fresh".into());
+            }
+        }
+        _ => return Err("initial optional owner presence differs".into()),
+    }
+    let record = p
+        .path(&["canonical", "world", "world", "organism_records"])?
+        .one()?;
+    record.path(&["state_graph", "brain", "content_digest"])?;
+    let mut omit = vec![
+        path(&["source_lineage"]),
+        path(&["policy"]),
+        path(&["canonical", "schema_version"]),
+        path(&["canonical", "residents", "0", "neural", "schema_version"]),
+        path(&[
+            "canonical",
+            "world",
+            "world",
+            "organism_records",
+            "0",
+            "state_graph",
+            "brain",
+            "content_digest",
+        ]),
+    ];
+    omit.push(path(&["controllers", owner, "optional_vocal"]));
+    let mut out = b"M2-INITIAL-RAW-PROJECTION-V1\0".to_vec();
+    project(p, &mut Vec::new(), &omit, &mut out)?;
+    Ok(out)
+}
+fn framed(bytes: &[u8], out: &mut Vec<u8>) {
+    out.extend_from_slice(&(bytes.len() as u64).to_le_bytes());
+    out.extend_from_slice(bytes);
+}
+fn project(
+    raw: Raw<'_>,
+    here: &mut Vec<String>,
+    omit: &[Vec<String>],
+    out: &mut Vec<u8>,
+) -> Result<()> {
+    // Preserve entire unaffected subtrees verbatim. Only containers on an exact
+    // exemption path need structural encoding, including the two resident arrays.
+    if !omit.iter().any(|p| p.starts_with(here.as_slice())) {
+        out.push(b'R');
+        framed(raw.0, out);
+        return Ok(());
+    }
+    match raw.0.first() {
+        Some(b'{') => {
+            out.push(b'O');
+            for (key, value) in raw.object()? {
+                here.push(key.clone());
+                if !omit.contains(here) {
+                    out.push(b'K');
+                    framed(key.as_bytes(), out);
+                    project(value, here, omit, out)?;
+                }
+                here.pop();
+            }
+            out.push(b'E');
+        }
+        Some(b'[') => {
+            out.push(b'A');
+            let items = raw.array()?;
+            out.extend_from_slice(&(items.len() as u64).to_le_bytes());
+            for (i, item) in items.into_iter().enumerate() {
+                here.push(i.to_string());
+                project(item, here, omit, out)?;
+                here.pop();
+            }
+        }
+        _ => return Err("projection path does not reach container".into()),
+    }
+    Ok(())
+}
+/// Exact hashes of every prohibited write bank and immutable source field.
+/// Dynamic activation, memory/predictor, sleep and biology are deliberately not frozen.
+pub fn closure(cp: &[u8]) -> Result<BTreeMap<String, String>> {
+    let p = payload(cp)?;
+    let mut out = BTreeMap::new();
+    for key in [
+        "schema_version",
+        "source_lineage",
+        "policy",
+        "runtime_profile_raw",
+        "selector_tag",
+        "decoder_stride",
+        "mode",
+        "source_inputs",
+        "recipe_digest",
+        "source_asset",
+        "founder",
+    ] {
+        out.insert(key.into(), hash(p.field(key)?.0));
+    }
+    out.insert(
+        "canonical.schema_version".into(),
+        hash(p.path(&["canonical", "schema_version"])?.0),
+    );
+    let resident = p.path(&["canonical", "residents"])?.one()?;
+    let neural = resident.field("neural")?;

+        "schema_version",
+        "organism_id",
+        "phenotype_hash",
+        "fast",
+        "lifetime",
+        "eligibility",
+        "weight_generation",

+        out.insert(format!("resident.neural.{key}"), hash(neural.field(key)?.0));
+    }
+    out.insert(
+        "resident.neural.dendrites".into(),
+        hash(neural.path(&["neural", "dendrites"])?.0),
+    );
+    out.insert(
+        "resident.neural_topology".into(),
+        hash(resident.field("neural_topology")?.0),
+    );

+        .path(&["canonical", "world", "world", "organism_records"])?
+        .one()?;

+        "genome",
+        "phenotype",
+        "birth_tick",
+        "organism_id",
+        "world_entity_id",
+    ] {
+        out.insert(format!("record.{key}"), hash(record.field(key)?.0));
+    }
+    for (owner, c) in p.field("controllers")?.object()? {
+        // Freeze optional ownership/schema, not its legitimate evolving last selection.
+        out.insert(
+            format!("controller.{owner}.optional_owner"),
+            match c.object()?.get("optional_vocal") {
+                Some(v) => hash(v.field("schema")?.0),
+                None => hash(b"absent"),

+        let motor = c.field("motor")?;
+        for key in [
+            "schema",
+            "source_inputs",
+            "source_hash",
+            "mode",
+            "m2_recipe",
+            "group_recipe",
+            "work_policy",
+        ] {
+            out.insert(
+                format!("controller.{owner}.{key}"),
+                hash(motor.field(key)?.0),
+            );
+        }
+    }
+    Ok(out)

+pub fn fresh(cp: &[u8], preparation: u64, sampler: u32) -> Result<()> {
+    // Both policies are Full in this research pair. Also proves the optional
+    // owner has exactly its fresh schema and null last-selection state.
+    matched_initial(
+        cp,
+        if cp.starts_with(OPTIONAL_MAGIC) {
+            "OptionalVocalM2/Full"
+        } else {
+            "LegacyGroupedM2/Full"
+        },
+    )?;
+    let p = payload(cp)?;
+    let canonical = p.field("canonical")?;
+    if canonical.field("sampling_seed")?.value::<u32>()? != sampler
+        || canonical.field("sampling_counter")?.value::<u32>()? != 0
+    {
+        return Err("fresh sampler differs".into());
+    }
+    let r = canonical.field("residents")?.one()?;
+    let n = r.field("neural")?;
+    if r.field("next_sequence")?.value::<u64>()? != 1
+        || !r.field("replay")?.array()?.is_empty()
+        || n.field("last_decision_tick")?.0 != b"null"
+        || n.field("last_outcome_tick")?.0 != b"null"
+        || n.field("dispatch_generation")?.value::<u64>()? != 0
+    {
+        return Err("fresh neural history differs".into());
+    }
+    for key in ["fast", "lifetime", "eligibility"] {
+        if n.field(key)?
+            .value::<Vec<f32>>()?
+            .iter()
+            .any(|v| v.to_bits() != 0)
+        {
+            return Err("fresh acquired bank is nonzero".into());
+        }
+    }
+    if canonical
+        .path(&["world", "world", "tick"])?
+        .value::<u64>()?
+        != preparation
+    {
+        return Err("fresh prepared tick differs".into());
+    }
+    Ok(())
+}

+/// In-memory comparison encoding only, NOT the checkpoint wire format. Every
