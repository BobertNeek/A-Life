+fn uncached_reference(
+    state: &GroupedMotorStateV1,
+    source: &BrainPhenotype,
+) -> Result<(), TrainingError> {
+    let expected = PhenotypeCompiler::compile_validated(
+        &state.source_inputs,
+        &BrainCapacityClass::n2048_grounded_drive_research(),
+    )?;
+    if state.schema != HOST_SCHEMA
+        || state.work_policy != GROUPED_HOST_WORK_POLICY_V1
+        || state.m2_recipe != M2_RECIPE_SHA256
+        || state.group_recipe != RECOVERY_GROUP_CONFIDENCE_RECIPE_SHA256
+        || state.mode == MediationModeV1::DirectSource
+        || expected != *source
+        || state.source_hash != source.phenotype_hash()
+        || source.neuron_count() != 2048
+        || source.synapses().len() != 34048
+        || source.foundation_abi().motor_lobe_m0().is_some()
+        || !source
+            .candidate_decoder()
+            .factorized_motor_channels(source)?
+            .contains(&MotorChannel::Posture)
+    {
+        return Err(invalid());
+    }
+    Lowered::pinned(MANIFEST)?;
+    GroupedMotorStateV1::group_recipe()?;
+    for (bank, count, bounded) in [
+        (state.activations.as_slice(), ADDED, false),
+        (state.ema.as_slice(), ADDED, true),
+        (state.metabolic.as_slice(), ADDED, true),
+        (state.group_activations.as_slice(), 16, false),
+        (state.group_ema.as_slice(), 16, true),
+        (state.group_metabolic.as_slice(), 16, true),
+    ] {
+        if bank.len() != count
+            || bank
+                .iter()
+                .any(|x| !x.is_finite() || (bounded && !(0. ..=1.).contains(x)))
+        {
+            return Err(invalid());
+        }
+    }
+    match &state.last_receipt {
+        None if state.generation == 0 => (),
+        Some(r) if state.generation > 0 => {
+            let core = &r.native.core;
+            if core.source_hash != state.source_hash
+                || core.mode != state.mode
+                || core.final_generation != state.generation
+                || core.source_generation.checked_add(1) != Some(state.generation)
+                || core.source_execution_count != 1
+                || core.appended_activations != state.activations
+                || r.native.extension_activations != state.group_activations
+                || r.native.schema != HOST_SCHEMA
+                || r.native.recipe != state.group_recipe
+                || core.schema != HOST_SCHEMA
+                || core.recipe != state.m2_recipe
+                || r.raw_memory.len() != core.components.len()
+                || r.raw_cognitive.len() != core.components.len()
+                || core.work != MediationWorkV1::expected()
+                || !r.applied_body_energy_debit.is_finite()
+                || r.applied_body_energy_debit < 0.
+            {
+                return Err(invalid());
+            }
+            finite(&r.raw_memory)?;
+            finite(&r.raw_cognitive)?;
+            r.native.scores.validate_host_identity(
+                state.source_hash,
+                r.frame_digest,
+                core.source_generation,
+            )?;
+        }
+        _ => return Err(invalid()),
+    }
+    Ok(())

+                .get(i as usize)
+                .copied()
+                .flatten(),
+            OptionalVocalChoiceV1::RestGroup => Some(4),
+            _ => None,
+        }
+    }
+    pub fn sample(
+        &self,
+        seed: u32,
+        counter: u32,
+        t: f64,
+    ) -> Result<OptionalVocalActionV1, TrainingError> {
+        let d = self.probabilities(t)?;
+        let mut draws = vec![0];
+        let representative = d.representative.sample(optional_draw(seed, counter, 0));
+        let mut motors = [OptionalVocalChoiceV1::NoOp; 6];
+        let mut op_factor = if representative == OptionalVocalChoiceV1::RestGroup {
+            Some(1)
+        } else {
+            None
+        };
+        let mut lp = d.representative.logp(representative)?;
+        for (s, motor) in motors.iter_mut().enumerate() {
+            if self.forced(representative) == Some(s) {
+                *motor = representative;
+            } else if let Some(f) = &d.channels[s] {
+                *motor = if f.outcomes == [OptionalVocalChoiceV1::NoOp] {
+                    OptionalVocalChoiceV1::NoOp
+                } else {
+                    let id = 2 + s as u32;
+                    draws.push(id);
+                    f.sample(optional_draw(seed, counter, id))
+                };
+                lp += f.logp(*motor)?;
+                if *motor == OptionalVocalChoiceV1::RestGroup {
+                    op_factor = Some(8);
+                }
+            }
+        }
+        let operation = if let Some(f) = op_factor {
+            draws.push(f);
+            let mask = self.native_binding.operation_mask();
+            let o = if mask[0]
+                && (!mask[1] || optional_draw(seed, counter, f) < d.conditional_operation[0])
+            {
+                0
+            } else {
+                1
+            };
+            lp += d.conditional_log_probability[o].ok_or_else(invalid)?;
+            Some(o as u8)
+        } else {
+            None
+        };
+        Ok(OptionalVocalActionV1 {
+            policy: OPTIONAL_VOCAL_POLICY_V1.into(),
+            binding_digest: self.digest()?,
+            representative,
+            motors,
+            operation,
+            temperature_bits: t.to_bits(),
+            joint_log_probability_bits: lp.to_bits(),
+            seed,
+            counter,
+            draw_factors: draws,
+        })
+    }
+    /// Closed evaluation verifies the actual RNG draw, not merely shape or logp.
+    /// No gradient is exposed. Future learning needs its own typed composition.
+    pub fn evaluate(
+        &self,
+        action: &OptionalVocalActionV1,
+    ) -> Result<OptionalVocalEvaluationV1, TrainingError> {
+        if *action != self.sample(action.seed, action.counter, action.temperature())? {
+            return Err(invalid());
+        }
+        let d = self.probabilities(action.temperature())?;
+        let mut factors = vec![d.representative.logp(action.representative)?];
+        for s in 0..6 {
+            if self.forced(action.representative) != Some(s) {
+                if let Some(f) = &d.channels[s] {
+                    factors.push(f.logp(action.motors[s])?);
+                }
+            }
+        }
+        if let Some(o) = action.operation {
+            factors.push(d.conditional_log_probability[o as usize].ok_or_else(invalid)?);
+        }
+        let mut physical = vec![];
+        for c in std::iter::once(action.representative).chain(action.motors) {
+            if let Some(i) = self.resolve(c, action.operation)? {
+                if !physical.contains(&i) {
+                    physical.push(i);
+                }
+            }
+        }
+        Ok(OptionalVocalEvaluationV1 {
+            joint_log_probability: action.joint_log_probability(),
+            joint_probability: action.joint_log_probability().exp(),
+            factor_log_probabilities: factors,
+            selected_physical_candidates_once: physical,
+        })
+    }
+    fn resolve(
+        &self,
+        c: OptionalVocalChoiceV1,
+        operation: Option<u8>,
+    ) -> Result<Option<u16>, TrainingError> {
+        match c {
+            OptionalVocalChoiceV1::Candidate(i) => Ok(Some(i)),
+            OptionalVocalChoiceV1::NoOp | OptionalVocalChoiceV1::Quiet => Ok(None),
+            OptionalVocalChoiceV1::RestGroup => operation
+                .and_then(|o| {
+                    self.native_binding
+                        .operations
+                        .get(o as usize)
+                        .copied()
+                        .flatten()
+                })
+                .map(|i| Some(i as u16))
+                .ok_or_else(invalid),
+        }
+    }
+    pub fn validate_sampling_context(
+        &self,
+        action: &OptionalVocalActionV1,
+        seed: u32,
+        counter: u32,
+        temperature: f64,
+    ) -> Result<(), TrainingError> {
+        if action.seed != seed
+            || action.counter != counter
+            || action.temperature_bits != temperature.to_bits()
+        {
+            return Err(invalid());
+        }
+        self.evaluate(action)?;
+        Ok(())
+    }
+    pub fn physical_selection(
+        &self,
+        action: &OptionalVocalActionV1,
+    ) -> Result<(u16, [Option<u16>; 6]), TrainingError> {
+        self.evaluate(action)?;
+        let representative = self
+            .resolve(action.representative, action.operation)?
+            .ok_or_else(invalid)?;
+        let mut motors = [None; 6];
+        for (m, c) in motors.iter_mut().zip(action.motors) {
+            *m = self.resolve(c, action.operation)?;
+        }
+        Ok((representative, motors))
+    }
+}
+fn optional_draw(seed: u32, counter: u32, factor: u32) -> f64 {
+    let mut v =
+        seed ^ RNG_DOMAIN ^ counter.wrapping_mul(0x9e3779b9) ^ factor.wrapping_mul(0x85ebca6b);
+    v = (v ^ (v >> 16)).wrapping_mul(0x7feb352d);
+    v = (v ^ (v >> 15)).wrapping_mul(0x846ca68b);
+    v ^= v >> 16;
+    (f64::from(v >> 9) + 0.5) / 8388608.
+}

+            || metadata["runtime_profile_raw"] != 5
+            || metadata["selector_tag"] != 12
+            || metadata["decoder_stride"] != 55
+            || metadata["recipe_digest"] != RECOVERY_GROUP_CONFIDENCE_RECIPE_SHA256

+            || !metadata["canonical"]["residents"]

+                .is_some_and(|rows| {
+                    !rows.is_empty()
+                        && rows.iter().all(|r| {
+                            r["neural"]["schema_version"]
+                                == crate::cpu::organism::OPTIONAL_VOCAL_NEURAL_SCHEMA
+                        })
+                })
+        {
+            return Err(CpuHostError::State("optional Vocal outer policy mismatch"));
+        }
+        let mut payload: GroupedPayloadV1 = serde_json::from_value(metadata.clone())?;
+        payload.restore_sampler_bits()?;
+        for state in payload.controllers.values() {
+            state.validate_policy::<OptionalVocalPolicy>()?;
